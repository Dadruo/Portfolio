sensor.ino · I2C temperature readsensor.ino · čtení teploty přes I2C
#include <Wire.h>

const uint8_t SENSOR_ADDR = 0x48;  // TMP102 on the I2C bus

float readTemperature() {
  Wire.beginTransmission(SENSOR_ADDR);
  Wire.write(0x00);                 // temperature register
  Wire.endTransmission();
  Wire.requestFrom(SENSOR_ADDR, (uint8_t)2);

  int16_t raw = ((Wire.read() << 8) | Wire.read()) >> 4;
  if (raw & 0x800) raw |= 0xF000;  // sign-extend 12-bit value
  return raw * 0.0625f;
}

void setup() { Serial.begin(115200); Wire.begin(); }
void loop()  { Serial.println(readTemperature()); delay(500); }
